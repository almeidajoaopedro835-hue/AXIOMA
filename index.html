Estrutura SQL (Supabase PostgreSQL) — Tabela de Registro de Clientes por Bilhete

CREATE TABLE public.client_tickets (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  client_registration_code VARCHAR(24) UNIQUE NOT NULL, -- Ex: CLI-ADM27-0001 (Código único para registro do cliente)
  ticket_security_hash TEXT UNIQUE NOT NULL,            -- Token único embutido no QR Code
  event_id UUID REFERENCES public.events(id),
  tier_id UUID REFERENCES public.ticket_tiers(id),
  client_full_name TEXT NOT NULL,
  client_cpf VARCHAR(14) NOT NULL,
  client_email TEXT NOT NULL,
  client_phone VARCHAR(20) NOT NULL,
  client_church TEXT,
  amount_paid NUMERIC(10,2) NOT NULL DEFAULT 0.00,
  payment_status VARCHAR(20) DEFAULT 'APPROVED',
  checked_in BOOLEAN DEFAULT FALSE,
  checked_in_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT now()
);
